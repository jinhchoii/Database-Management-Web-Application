
-- Drop and recreate the database
DROP DATABASE IF EXISTS ConferenceDB;
CREATE DATABASE ConferenceDB;
USE ConferenceDB;

-- Table creation
CREATE TABLE Committee (
    committee_id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);

CREATE TABLE CommitteeMember (
    member_id INT PRIMARY KEY,
    name VARCHAR(100),
    email VARCHAR(100),
    committee_id INT,
    FOREIGN KEY (committee_id) REFERENCES Committee(committee_id)
);

CREATE TABLE Conference (
    conference_id INT PRIMARY KEY,
    name VARCHAR(100)
);

CREATE TABLE ConferenceRoom (
    room_id INT PRIMARY KEY,
    room_number VARCHAR(50),
    capacity INT
);

CREATE TABLE HotelRoom (
    room_id INT PRIMARY KEY,
    hotel_name VARCHAR(100),
    room_number VARCHAR(20),
    capacity INT
);

CREATE TABLE Student (
    student_id INT PRIMARY KEY,
    name VARCHAR(100),
    department VARCHAR(100),
    student_number VARCHAR(20)
);

CREATE TABLE ProfessionalAttendee (
    pro_id INT PRIMARY KEY,
    name VARCHAR(100),
    email VARCHAR(100)
);

CREATE TABLE Registration (
    reg_id INT PRIMARY KEY,
    student_id INT,
    pro_id INT,
    amount DECIMAL(10,2),
    FOREIGN KEY (student_id) REFERENCES Student(student_id),
    FOREIGN KEY (pro_id) REFERENCES ProfessionalAttendee(pro_id)
);

CREATE TABLE RoomAssignment (
    assignment_id INT PRIMARY KEY,
    student_id INT,
    hotel_room_id INT,
    FOREIGN KEY (student_id) REFERENCES Student(student_id),
    FOREIGN KEY (hotel_room_id) REFERENCES HotelRoom(room_id)
);

CREATE TABLE Session (
    session_id INT PRIMARY KEY,
    conference_id INT,
    name VARCHAR(100),
    date DATE,
    time TIME,
    start_end VARCHAR(50),
    location VARCHAR(100),
    room_id INT,
    FOREIGN KEY (conference_id) REFERENCES Conference(conference_id),
    FOREIGN KEY (room_id) REFERENCES ConferenceRoom(room_id)
);

CREATE TABLE Speaker (
    speaker_id INT PRIMARY KEY,
    name VARCHAR(100)
);

CREATE TABLE SessionSpeaker (
    session_id INT,
    speaker_id INT,
    PRIMARY KEY (session_id, speaker_id),
    FOREIGN KEY (session_id) REFERENCES Session(session_id),
    FOREIGN KEY (speaker_id) REFERENCES Speaker(speaker_id)
);

CREATE TABLE Sponsor (
    sponsor_id INT PRIMARY KEY,
    company_name VARCHAR(100),
    level VARCHAR(50)
);

CREATE TABLE Job (
    job_id INT PRIMARY KEY,
    title VARCHAR(100),
    description TEXT,
    sponsor_id INT,
    FOREIGN KEY (sponsor_id) REFERENCES Sponsor(sponsor_id)
);

CREATE TABLE Sponsorship (
    sponsor_id INT,
    conference_id INT,
    amount DECIMAL(10,2),
    PRIMARY KEY (sponsor_id, conference_id),
    FOREIGN KEY (sponsor_id) REFERENCES Sponsor(sponsor_id),
    FOREIGN KEY (conference_id) REFERENCES Conference(conference_id)
);
